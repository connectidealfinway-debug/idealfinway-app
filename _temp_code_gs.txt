function doGet(e) {
  try {
    var action = e && e.parameter && e.parameter.action;
    var result;
    if      (action === 'getData')      result = getData();
    else if (action === 'getEmployees') result = getEmployees();
    else if (action === 'fixjh') { result = setParentIdsForReferrals(); }
    else if (action === 'checkTargets') { result = checkTradeTargets(); }
    else if (action === 'sendCallSummary') { result = sendDailyCallTradeSummary(); }
    else if (action === 'migrateOfficeBrok') { result = migrateOrphanBrokerageToOffice(); }
    else if (action === 'lastUpdate') { result = {success:true, ts: PropertiesService.getScriptProperties().getProperty('lastUpdate')||'0'}; }
    else result = {status:'ok', message:'IdealFinway API v9 Running'};
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.TEXT);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({success:false, error:err.toString()})).setMimeType(ContentService.MimeType.TEXT);
  }
}

function getSheet(name) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sheet = ss.getSheetByName(name);
  if (!sheet) { sheet = ss.insertSheet(name); }
  if (sheet.getLastRow() === 0) {
    if (HEADERS[name]) sheet.appendRow(HEADERS[name]);
  } else {
    var firstRow = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0];
    var expected = HEADERS[name];
    if (expected && firstRow[0] !== expected[0]) {
      sheet.insertRowBefore(1);
      sheet.getRange(1,1,1,expected.length).setValues([expected]);
    } else if (expected && firstRow.length < expected.length) {
      // naye columns (jaise Employee me Commission) header me add karo
      sheet.getRange(1,1,1,expected.length).setValues([expected]);
    }
  }
  return sheet;
}

var TZ = 'GMT+5:30';
function normDate(val) {
  if (val instanceof Date) return Utilities.formatDate(val, TZ, 'yyyy-MM-dd');
  if (typeof val === 'string' && val.length > 10) {
    var pd = new Date(val);
    if (!isNaN(pd.getTime())) return Utilities.formatDate(pd, TZ, 'yyyy-MM-dd');
  }
  return val;
}
function toYMD(v){
  if (v instanceof Date) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
  var s = String(v); var d = new Date(s);
  if (!isNaN(d.getTime()) && s.length >= 8) return Utilities.formatDate(d, TZ, 'yyyy-MM-dd');
  return s;
}

function sheetToObjects(name) {
  var sheet = getSheet(name);
  var last = sheet.getLastRow();
  if (last < 2) return [];
  var headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0];
  var data = sheet.getRange(2,1,last-1,sheet.getLastColumn()).getValues();
  var result = [];
  for (var i = 0; i < data.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      var val = data[i][j];
      var hn = String(headers[j]).toLowerCase();
      if (hn.indexOf('time') !== -1) {
        if (val instanceof Date) val = ('0'+val.getHours()).slice(-2)+':'+('0'+val.getMinutes()).slice(-2);
      } else if (hn === 'month') {
        if (val instanceof Date) val = Utilities.formatDate(val, TZ, 'yyyy-MM');
      } else if (hn.indexOf('date') !== -1 || hn.indexOf('meeting') !== -1 || hn.indexOf('follow') !== -1) {
        val = normDate(val);
      } else if (val instanceof Date) {
        val = Utilities.formatDate(val, TZ, 'yyyy-MM-dd');
      }
      row[headers[j]] = val !== undefined && val !== null ? String(val) : '';
    }
    row['_rowNum'] = i + 2;
    result.push(row);
  }
  return result;
}

// ===== CLIENT BROKERAGE (Choice Bi Report upload) =====

function addClientBrokerageBatch(data) {
  try {
    var rows = data.rows || [];
    if (!rows.length) return {success:false, error:'No rows'};
    var sheet = getSheet('ClientBrokerage');
    var out = [];
    for (var i = 0; i < rows.length; i++) {
      var r = rows[i];
      out.push([
        r['Date']||'', r['From Date']||'', r['To Date']||'',
        r['Client ID']||'', r['Client Name']||'',
        r['Employee ID']||'', r['Employee Name']||'', r['Branch']||'',
        parseFloat(r['Amount'])||0, parseFloat(r['Total Brokerage'])||0,
        r['Upload ID']||''
      ]);
    }
    sheet.getRange(sheet.getLastRow()+1, 1, out.length, out[0].length).setValues(out);
    return {success:true, added: out.length};
  } catch(err) { return {success:false, error:err.toString()}; }
}

function deleteClientBrokerageRange(data) {
  try {
    var from = String(data.fromDate||''), to = String(data.toDate||'');
    if (!from || !to) return {success:false, error:'fromDate/toDate required'};
    var sheet = getSheet('ClientBrokerage');
    var last = sheet.getLastRow();
    if (last < 2) return {success:true, deleted:0};
    var vals = sheet.getRange(2,1,last-1,3).getValues(); // Date, From Date, To Date
    var del = [];
    for (var i = 0; i < vals.length; i++) {
      if (toYMD(vals[i][1]) === from && toYMD(vals[i][2]) === to) del.push(i+2);
    }
    // neeche se upar delete karo taki row numbers shift na ho
    for (var j = del.length-1; j >= 0; j--) sheet.deleteRow(del[j]);
    return {success:true, deleted: del.length};
  } catch(err) { return {success:false, error:err.toString()}; }
}

// ===== MONTHLY HISAB (upsert by Month + Employee ID) =====

function saveHisab(data) {
  try {
    var sheet = getSheet('Hisab');
    var month = String(data['Month']||'');
    var eid = String(data['Employee ID']||'');
    if (!month || !eid) return {success:false, error:'Month & Employee ID required'};
    var last = sheet.getLastRow();
    var found = 0;
    if (last >= 2) {
      var vals = sheet.getRange(2,1,last-1,2).getValues();
      for (var i = 0; i < vals.length; i++) {
        var m = vals[i][0];
        if (m instanceof Date) m = Utilities.formatDate(m, TZ, 'yyyy-MM');
        if (String(m) === month && String(vals[i][1]) === eid) { found = i+2; break; }
      }
    }
    var rowVals = [
      month, eid, data['Employee Name']||'', data['Branch']||'',
      parseFloat(data['Brokerage'])||0, parseFloat(data['Commission %'])||0,
      parseFloat(data['Brok Share'])||0, parseFloat(data['Additions'])||0,
      parseFloat(data['Deductions'])||0, parseFloat(data['Net Payable'])||0,
      data['Status']||'Pending', data['Paid Date']||'', data['Details']||''
    ];
    if (found) {
      sheet.getRange(found,1,1,rowVals.length).setValues([rowVals]);
      // Month column ko text रखो taki '2026-07' date me convert na ho
      sheet.getRange(found,1).setNumberFormat('@').setValue(month);
    } else {
      sheet.appendRow(rowVals);
      sheet.getRange(sheet.getLastRow(),1).setNumberFormat('@').setValue(month);
    }
    return {success:true};
  } catch(err) { return {success:false, error:err.toString()}; }
}

function punchAttendance(data) {
  try {
    var sheet = getSheet('Attendance');
    var last = sheet.getLastRow();
    var targetDate = String(data['Date']||'');
    var eid = String(data['Employee ID']||'');
    var found = 0;
    if (last >= 2) {
      var vals = sheet.getRange(2,1,last-1,sheet.getLastColumn()).getValues();
      for (var i=0;i<vals.length;i++){
        if (toYMD(vals[i][0]) === toYMD(targetDate) && String(vals[i][1]) === eid) { found = i+2; break; }
      }
    }
    if (!found) {
      sheet.appendRow([targetDate,eid,data['Employee Name']||'',data['Branch']||'','','',data['Status']||'Present',data['Remarks']||'']);
      found = sheet.getLastRow();
    }
    if (data['In Time'] !== undefined && data['In Time'] !== '') { var ci = sheet.getRange(found,5); ci.setNumberFormat('@'); ci.setValue(String(data['In Time'])); }
    if (data['Out Time'] !== undefined && data['Out Time'] !== '') { var co = sheet.getRange(found,6); co.setNumberFormat('@'); co.setValue(String(data['Out Time'])); }

// ===== Employee offboarding: auto-move client data to Office, resignation approval workflow, rejoin/restore =====
var OFFICE_EMP_ID = 'ideal.office';
var OFFICE_EMP_NAME = 'Office client';
var REASSIGN_SHEETS = ['Accounts','SIP','Insurance','Students','ClientBrokerage','Leads','Meetings'];

function reassignEmployeeDataToOffice(empId, empName) {
  var counts = {};
  var today = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
  var moveSheet = getSheet('EmployeeDataMoves');
  REASSIGN_SHEETS.forEach(function(sheetName){
    var sheet = getSheet(sheetName);
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) { counts[sheetName]=0; return; }
    var lastCol = sheet.getLastColumn();
    var data = sheet.getRange(1,1,lastRow,lastCol).getValues();
    var headers = data[0];
    var empIdCol = headers.indexOf('Employee ID');
    if (empIdCol === -1) { counts[sheetName]=0; return; }
    var empNameCol = headers.indexOf('Employee Name');
    var noteCol = headers.indexOf('Remarks');
    if (noteCol === -1) noteCol = headers.indexOf('Reference');
    if (noteCol === -1) noteCol = headers.indexOf('Reference By');
    var clientIdCol = headers.indexOf('Client ID');
    var clientNameCol = headers.indexOf('Client Name');
    if (clientNameCol === -1) clientNameCol = headers.indexOf('Student Name');
    var n = 0;
    for (var i=1;i<data.length;i++){
      if (String(data[i][empIdCol]).trim() === String(empId).trim()){
        var rowNum = i+1;
        moveSheet.appendRow([today, sheetName, rowNum, clientIdCol>=0?data[i][clientIdCol]:'', clientNameCol>=0?data[i][clientNameCol]:'', empId, empName, OFFICE_EMP_ID, OFFICE_EMP_NAME, 'Moved', '']);
        sheet.getRange(rowNum, empIdCol+1).setValue(OFFICE_EMP_ID);
        if (empNameCol>=0) sheet.getRange(rowNum, empNameCol+1).setValue(OFFICE_EMP_NAME);
        if (noteCol>=0){
          var existingNote = data[i][noteCol] || '';
          var tag = '[Ex-employee: '+empName+']';
          if (String(existingNote).indexOf(tag)===-1){
            sheet.getRange(rowNum, noteCol+1).setValue((existingNote?existingNote+' ':'')+tag);
          }
        }
        n++;
      }
    }
    counts[sheetName]=n;
  });
  return counts;
}

function rejoinEmployee(data) {
  try {
    var empId = data['Employee ID'];
    if (!empId) return {success:false, error:'Missing Employee ID'};
    var empSheet = getSheet('Employee');
    var empData = empSheet.getDataRange().getValues();
    var empHeaders = empData[0];
    var idCol = empHeaders.indexOf('Employee ID');
    var nameCol = empHeaders.indexOf('Name');
    var statusCol = empHeaders.indexOf('Status');
    var empRowNum = -1, empName = '';
    for (var i=1;i<empData.length;i++){
      if (String(empData[i][idCol]).trim() === String(empId).trim()){ empRowNum = i+1; empName = empData[i][nameCol]; break; }
    }
    if (empRowNum === -1) return {success:false, error:'Employee not found'};
    empSheet.getRange(empRowNum, statusCol+1).setValue('Active');
    var moveSheet = getSheet('EmployeeDataMoves');
    var moveData = moveSheet.getDataRange().getValues();
    var restored = {};
    if (moveData.length > 1) {
      var mh = moveData[0];
      var fromIdCol = mh.indexOf('From Employee ID');
      var sheetCol = mh.indexOf('Sheet');
      var rowNumCol = mh.indexOf('RowNum');
      var statusColM = mh.indexOf('Status');
      var restoredDateCol = mh.indexOf('Restored Date');
      for (var j=1;j<moveData.length;j++){
        if (String(moveData[j][fromIdCol]).trim() === String(empId).trim() && moveData[j][statusColM] === 'Moved'){
          var sName = moveData[j][sheetCol];
          var rNum = moveData[j][rowNumCol];
          var targetSheet = getSheet(sName);
          var tHeaders = targetSheet.getRange(1,1,1,targetSheet.getLastColumn()).getValues()[0];
          var eIdCol = tHeaders.indexOf('Employee ID');
          var eNameCol = tHeaders.indexOf('Employee Name');
          if (eIdCol>=0){
            var curVal = targetSheet.getRange(rNum, eIdCol+1).getValue();
            if (String(curVal).trim() === OFFICE_EMP_ID){
              targetSheet.getRange(rNum, eIdCol+1).setValue(empId);
              if (eNameCol>=0) targetSheet.getRange(rNum, eNameCol+1).setValue(empName);
              restored[sName] = (restored[sName]||0)+1;
              moveSheet.getRange(j+1, statusColM+1).setValue('Restored');
              if (restoredDateCol>=0) moveSheet.getRange(j+1, restoredDateCol+1).setValue(Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd'));
            }
          }
        }
      }
    }
    touchLastUpdate();
    return {success:true, restored:restored};
  } catch(err) { return {success:false, error:err.toString()}; }
}

function numToWords(n) {
  n = Math.round(parseFloat(n)||0);
  if (n === 0) return 'Zero';
  var neg = n < 0; if (neg) n = -n;
  var ones = ['','One','Two','Three','Four','Five','Six','Seven','Eight','Nine','Ten','Eleven','Twelve','Thirteen','Fourteen','Fifteen','Sixteen','Seventeen','Eighteen','Nineteen'];
  var tens = ['','','Twenty','Thirty','Forty','Fifty','Sixty','Seventy','Eighty','Ninety'];
  function two(x){ return x < 20 ? ones[x] : tens[Math.floor(x/10)] + (x%10 ? ' ' + ones[x%10] : ''); }
  function three(x){ return (Math.floor(x/100) ? ones[Math.floor(x/100)] + ' Hundred' + (x%100 ? ' ' : '') : '') + (x%100 ? two(x%100) : ''); }
  var out = '';
  var crore = Math.floor(n/10000000); n %= 10000000;
  var lakh = Math.floor(n/100000); n %= 100000;
  var thou = Math.floor(n/1000); n %= 1000;
  if (crore) out += three(crore) + ' Crore ';
  if (lakh) out += two(lakh) + ' Lakh ';
  if (thou) out += two(thou) + ' Thousand ';
  if (n) out += three(n);
  return (neg ? 'Minus ' : '') + out.replace(/s+/g,' ').trim();
}

// ===== v9: DAILY AUTOMATED BACKUP — poori spreadsheet ki daily copy alag folder me =====
// permanent data-safety solution: agar live sheet me galti se data delete/corrupt ho jaye,
// har din ki poori copy "IdealFinway Backups" folder me available rehti h. 45 din se purani
// copies auto-delete hoti h (Drive storage bachane ke liye) — is number ko yahan badal sakte h.
function dailyBackup() {
  try {
    var file = DriveApp.getFileById(SHEET_ID);
    var folderName = 'IdealFinway Backups';
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    var ts = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd_HH-mm');
    file.makeCopy('IdealFinway Backup ' + ts, folder);
    var cutoff = new Date(); cutoff.setDate(cutoff.getDate() - 45);
    var files = folder.getFiles();
    while (files.hasNext()) {
      var f = files.next();
      if (f.getDateCreated() < cutoff) f.setTrashed(true);
    }
  } catch(err) {
    // backup fail hone se live app kabhi na tute — silent fail, bas email kar do admin ko
    try { MailApp.sendEmail({to:'connectidealfinway@gmail.com', subject:'IdealFinway Backup Failed', body:String(err)}); } catch(e2) {}
  }
}

function installDailyBackupTrigger() {
  ScriptApp.getProjectTriggers().forEach(function(t){
    if (t.getHandlerFunction() === 'dailyBackup') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('dailyBackup').timeBased().everyDays(1).atHour(2).create();

}

function sendWhatsAppMessage(toPhone, message) {
  try {
    var props = PropertiesService.getScriptProperties();
    var token = props.getProperty('WHATSAPP_TOKEN');
    var phoneId = props.getProperty('WHATSAPP_PHONE_ID');
    if (!token || !phoneId || !toPhone) return {success:false, skipped:'not configured'};
    var url = 'https://graph.facebook.com/v20.0/' + phoneId + '/messages';
    var payload = { messaging_product:'whatsapp', to: String(toPhone).replace(/[^0-9]/g,''), type:'text', text:{ body: message } };
    var res = UrlFetchApp.fetch(url, {
      method:'post', contentType:'application/json',
      headers:{ Authorization:'Bearer ' + token },
      payload: JSON.stringify(payload), muteHttpExceptions:true
    });
    return {success:true, status:res.getResponseCode()};
  } catch(err) { return {success:false, error:err.toString()}; }
}

function sendDailyCallTradeSummary() {
  try {
    var today = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
    var rows = sheetToObjects('CallTrade').filter(function(r){ return normDate(r['Date']) === today; });
    if (!rows.length) return {success:true, sent:0, note:'no calls logged today'};
    var byEmp = {};
    rows.forEach(function(r){
      var id = r['Employee ID']||'unknown';
      if (!byEmp[id]) byEmp[id] = { name: r['Employee Name']||id, calls:0, trades:0 };
      byEmp[id].calls++;
      if (r['Trade Done']==='Yes') byEmp[id].trades++;
    });
    var employees = sheetToObjects('Employee');
    var empMap = {};
    employees.forEach(function(e){ empMap[e['Employee ID']] = e; });
    var adminPhone = PropertiesService.getScriptProperties().getProperty('ADMIN_WHATSAPP');
    var results = [];
    Object.keys(byEmp).forEach(function(empId){
      var stat = byEmp[empId];
      var emp = empMap[empId];
      var msg = 'IdealFinway Daily Update (' + today + ')\n' + stat.name + ': ' + stat.calls + ' calls, ' + stat.trades + ' trades done today.';
      if (adminPhone) sendWhatsAppMessage(adminPhone, msg);
      if (emp && emp['Parent ID']) {
        var parent = empMap[emp['Parent ID']];
        if (parent && parent['WhatsApp']) sendWhatsAppMessage(parent['WhatsApp'], msg);
      }
      results.push({employee:empId, calls:stat.calls, trades:stat.trades});
    });
    return {success:true, sent:results.length, summary:results};
  } catch(err) { return {success:false, error:err.toString()}; }
}

function checkTradeTargets() {
  try {
    var now = new Date();
    var ist = Utilities.formatDate(now, TZ, 'HH:mm');
    var day = Utilities.formatDate(now, TZ, 'EEE');
    if (['Sat','Sun'].indexOf(day) !== -1) return {success:true, skipped:'weekend'};
    if (ist < '09:15' || ist > '15:30') return {success:true, skipped:'outside market hours'};

    var trades = sheetToObjects('TradingJournal');
    var openTrades = trades.filter(function(t){ return t['Status']==='Open' || t['Status']==='1st Target Done'; });
    if (!openTrades.length) return {success:true, checked:0};

    var symbols = [];
    openTrades.forEach(function(t){ var s=(t['Stock']||'').trim().toUpperCase(); if(s && symbols.indexOf(s)===-1) symbols.push(s); });

    var ss = SpreadsheetApp.openById(SHEET_ID);
    var helper = ss.getSheetByName('_PriceHelper');
    if (!helper) helper = ss.insertSheet('_PriceHelper');
    helper.getRange(1,1,Math.max(helper.getLastRow(),symbols.length,1),2).clearContent();
    symbols.forEach(function(sym,i){
      helper.getRange(i+1,1).setValue(sym);
      helper.getRange(i+1,2).setFormula('=IFERROR(GOOGLEFINANCE("NSE:'+sym+'"),"NA")');
    });
    SpreadsheetApp.flush();
    Utilities.sleep(3000);
    var priceMap = {};
    var vals = helper.getRange(1,1,symbols.length,2).getValues();
    vals.forEach(function(r){ priceMap[r[0]] = r[1]; });

    var sheet = getSheet('TradingJournal');
    var headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0];
    var colIdx = {}; headers.forEach(function(h,i){ colIdx[h]=i+1; });
    var updated = 0;

    openTrades.forEach(function(t){
      var sym = (t['Stock']||'').trim().toUpperCase();
      var price = priceMap[sym];
      if (price===undefined || price==='NA' || price==='' || isNaN(price)) return;
      price = parseFloat(price);
      var dir = t['Direction'];
      var sl = parseFloat(t['SL']);
      var t1 = parseFloat(t['Target1']);
      var t2 = t['Target2'] ? parseFloat(t['Target2']) : null;
      var newStatus = null, exitPrice = null;

      if (dir === 'Buy') {
        if (t['Status']==='Open') {
          if (!isNaN(sl) && price <= sl) { newStatus='SL Done'; exitPrice=sl; }
          else if (t2 && price >= t2) { newStatus='2nd Target Done'; exitPrice=t2; }
          else if (!isNaN(t1) && price >= t1) {
            if (t2) { newStatus='1st Target Done'; } else { newStatus='Target Done'; exitPrice=t1; }
          }
        } else if (t['Status']==='1st Target Done') {
          if (t2 && price >= t2) { newStatus='2nd Target Done'; exitPrice=t2; }
          else if (!isNaN(t1) && price <= t1) { newStatus='Target Done'; exitPrice=t1; }
        }
      } else {
        if (t['Status']==='Open') {
          if (!isNaN(sl) && price >= sl) { newStatus='SL Done'; exitPrice=sl; }
          else if (t2 && price <= t2) { newStatus='2nd Target Done'; exitPrice=t2; }
          else if (!isNaN(t1) && price <= t1) {
            if (t2) { newStatus='1st Target Done'; } else { newStatus='Target Done'; exitPrice=t1; }
          }
        } else if (t['Status']==='1st Target Done') {
          if (t2 && price <= t2) { newStatus='2nd Target Done'; exitPrice=t2; }
          else if (!isNaN(t1) && price >= t1) { newStatus='Target Done'; exitPrice=t1; }
        }
      }

      if (newStatus) {
        var r = t['_rowNum'];
        sheet.getRange(r, colIdx['Status']).setValue(newStatus);
        if (exitPrice !== null) {
          var qty = parseFloat(t['Qty'])||0;
          var entry = parseFloat(t['Entry Price'])||0;
          var result = (exitPrice - entry) * qty * (dir==='Buy'?1:-1);
          var resultPct = entry ? (((exitPrice-entry)/entry)*100*(dir==='Buy'?1:-1)) : 0;
          sheet.getRange(r, colIdx['Exit Price']).setValue(exitPrice);
          sheet.getRange(r, colIdx['Exit Date']).setValue(Utilities.formatDate(now, TZ, 'yyyy-MM-dd'));
          sheet.getRange(r, colIdx['Result']).setValue(Math.round(result*100)/100);
          sheet.getRange(r, colIdx['Result %']).setValue(Math.round(resultPct*100)/100);
        }
        updated++;
      }
    });

    return {success:true, checked:openTrades.length, updated:updated};
  } catch(err) { return {success:false, error:err.toString()}; }
}

function uploadClientDocument(data){
  try{
    var rootFolders=DriveApp.getFoldersByName('IdealFinway Client Documents');
    var rootFolder=rootFolders.hasNext()?rootFolders.next():DriveApp.createFolder('IdealFinway Client Documents');
    var cid=(data['Client ID']||'unknown').toString();
    var subFolders=rootFolder.getFoldersByName(cid);
    var clientFolder=subFolders.hasNext()?subFolders.next():rootFolder.createFolder(cid);
    var bytes=Utilities.base64Decode(data['FileData']);
    var blob=Utilities.newBlob(bytes,data['MimeType']||'application/octet-stream',data['FileName']||'document');
    var file=clientFolder.createFile(blob);
    try{file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);}catch(e2){}
    getSheet('ClientDocuments').appendRow([new Date(),cid,data['Client Name']||'',data['Doc Type']||'Other',data['FileName']||'',file.getId(),file.getUrl(),data['Uploaded By']||'',data['Employee ID']||'']);
    return {success:true, fileId:file.getId(), url:file.getUrl()};
  }catch(err){ return {success:false, error:err.toString()}; }
}

// Create a new named IPO with its three category (Retail/sHNI/bHNI) listing-gain values.
function addIPOMaster(data) {

// Bulk-add N individual application placeholder rows (1 PAN card = 1 application) sharing
// Reference/Category/Broker/Buy_Rate - PAN/Client_Name/Mobile stay blank, filled in one at a
// time later via editRow, exactly like the spec: "bulk-adding N applications creates N
// individual placeholder rows sharing reference/category/broker/buy rate".
function addIPOApplications(data) {
  try {
    var ipoId = String(data['IPO_ID']||'');
    var qty = parseInt(data['Qty'], 10) || 0;
    if (!ipoId) return {success:false, error:'IPO_ID required'};
    if (qty < 1) return {success:false, error:'Qty must be at least 1'};
    var sheet = getSheet('IPO_Applications');
    var now = normDate(new Date());
    var buyRate = parseFloat(data['Buy_Rate'])||0;
    var category = data['Category']||'Retail';
    var broker = data['Broker']||'Choice';
    var reference = data['Reference']||'';
    var out = [];
    var baseTs = new Date().getTime();
    for (var i = 0; i < qty; i++) {
      var appId = 'APP' + baseTs + '_' + i;
      out.push([
        appId, ipoId, reference, category, broker, buyRate,
        '', '', '',            // PAN, Client_Name, Mobile - filled in later
        'Pending', 0,           // Status, Penalty
        'No', '', '',           // Payment_Done, Payment_Account, Payment_By
        'No', '', '', 'No', '', '', // Resold, Resale_Buyer, Resale_Rate, Resale_Payment_Done, Resale_Account, Resale_By
        now, now,                // Created_Date, Updated_Date
        ''                        // Client_ID - filled in later (Choice broker demat client ID)
      ]);
    }
    sheet.getRange(sheet.getLastRow()+1, 1, out.length, out[0].length).setValues(out);
    return {success:true, added: out.length};
  } catch(err) { return {success:false, error:err.toString()}; }
}

// Bulk resale action: buyer name + resale rate + quantity -> marks that many not-yet-resold
// applications (rowNums picked by the frontend from the current filtered view) as resold.
// Client settlement stays the full buy rate always once resold (status control is disabled
// client-side for resold rows), company profit = Resale_Rate - Buy_Rate computed on read.
function bulkResellIPOApplications(data) {
  try {
    var rowNums = data['rowNums']||[];
    var buyer = String(data['Resale_Buyer']||'').trim();
    var rate = parseFloat(data['Resale_Rate']);
    if (!rowNums.length) return {success:false, error:'No applications selected'};
    if (!buyer) return {success:false, error:'Resale_Buyer required'};
    if (isNaN(rate)) return {success:false, error:'Resale_Rate required'};
    var sheet = getSheet('IPO_Applications');
    var headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0];
    var now = normDate(new Date());
    var vals = {'Resold':'Yes','Resale_Buyer':buyer,'Resale_Rate':rate,'Updated_Date':now};
    var updated = 0;
    rowNums.forEach(function(rn){
      rn = parseInt(rn,10);
      if (!rn || rn < 2) return;
      for (var key in vals) {
        var idx = headers.indexOf(key);
        if (idx >= 0) sheet.getRange(rn, idx+1).setValue(vals[key]);
      }
      updated++;
    });
    return {success:true, updated: updated};
  } catch(err) { return {success:false, error:err.toString()}; }
}
